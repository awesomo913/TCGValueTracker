- [ ] **Step 3: implement `app_mode_acceptance.py`:**
```python
"""Honest ship-gate for App Mode: real gradle build if the Android toolchain is
present, otherwise a structure/manifest check that NEVER silently passes nothing."""
from __future__ import annotations

import os
import shutil
from pathlib import Path


def _android_toolchain_present(project_dir: str) -> bool:
    has_gradlew = (Path(project_dir) / "gradlew").exists() or \
                  (Path(project_dir) / "gradlew.bat").exists()
    has_gradle = shutil.which("gradle") is not None
    has_sdk = bool(os.environ.get("ANDROID_HOME") or os.environ.get("ANDROID_SDK_ROOT"))
    return (has_gradlew or has_gradle) and has_sdk


def acceptance_kind(project_dir: str) -> str:
    return "gradle" if _android_toolchain_present(project_dir) else "structure"


def gradle_command(project_dir: str) -> list[str]:
    gw = Path(project_dir) / ("gradlew.bat" if os.name == "nt" else "gradlew")
    exe = str(gw) if gw.exists() else "gradle"
    return [exe, ":app:assembleDebug", "--console=plain"]


def structure_check(project_dir: str) -> tuple[bool, list[str]]:
    root = Path(project_dir)
    reasons: list[str] = []

    manifests = list(root.rglob("AndroidManifest.xml"))
    if not manifests:
        reasons.append("missing AndroidManifest.xml")
    gradles = list(root.rglob("build.gradle")) + list(root.rglob("build.gradle.kts"))
    if not gradles:
        reasons.append("missing build.gradle(.kts)")
    else:
        gtext = " ".join(g.read_text(encoding="utf-8", errors="ignore") for g in gradles)
        if "applicationId" not in gtext:
            reasons.append("build.gradle missing applicationId")
        if "signingConfig" not in gtext and "signingConfigs" not in gtext:
            reasons.append("build.gradle missing signing config")
    if manifests:
        mtext = " ".join(m.read_text(encoding="utf-8", errors="ignore") for m in manifests)
        if "ads.APPLICATION_ID" not in mtext and "applovin" not in mtext.lower():
            reasons.append("manifest missing AdMob APPLICATION_ID")
    if not list(root.rglob("mipmap*/ic_launcher*")):
        reasons.append("missing launcher icon (res/mipmap*/ic_launcher)")

    return (len(reasons) == 0, reasons)
```
